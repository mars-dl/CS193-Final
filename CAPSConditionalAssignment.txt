#include <iostream>
#include <limits>
using namespace std;
/*Ethan Ma
Conditional Assignment 3/2/26*/

int main() {
    bool seeLoop = 1;
    while (seeLoop) {

        int i;
        int j;

        while ((cout << "Enter an integer:" << endl)
         && !(cin >> i)) {
            cout << "Invalid input; ";
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
         }
        while ((cout << "Enter another integer:" << endl)
         && !(cin >> j)) {
            cout << "Invalid input; ";
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
         }
        int sum = i+j;
        if (sum > 0) {
            cout << "The sum of " << i << " and " << j << " is positive" << endl;
        }
        else if (sum < 0) {
            cout << "The sum of " << i << " and " << j << " is negative" << endl;
        }
        else {
            cout << "The sum of " << i << " and " << j << " is zero" << endl;
        }

        while ((cout << "Would you like to repeat? 1 for yes, 0 for no:" << endl)
         && !(cin >> seeLoop)) {
            cout << "Invalid input; ";
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
         }
    }

    //simple calculator
    seeLoop = 1;
    float k;
    float l;
    string operation;
    while (seeLoop) {
         while ((cout << endl << "Enter an integer:" << endl)
         && !(cin >> k)){
            cout << "Invalid input; ";
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
         }
         while (!(operation == "+" || operation == "-" || operation == "*" || operation == "/")) {
            cout << "Enter an operation (+, -, *, /):" << endl;
            cin >> operation;
         }
        while ((cout << "Enter another integer:" << endl)
         && !(cin >> l)) {
            cout << "Invalid input; ";
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
         }
        if (operation == "+")
            cout << k << " + " << l << " = " << k + l << endl;
        else if (operation == "-")
            cout << k << " - " << l << " = " << k - l << endl;
        else if (operation == "*")
            cout << k << " * " << l << " = " << k * l << endl;
        else if (operation == "/" && l != 0)
            cout << k << " / " << l << " = " << k / l << endl;
        else {
            cout << "Error" << endl;
        }
        operation = "";
        while ((cout << "Would you like to repeat? 1 for yes, 0 for no:" << endl)
         && !(cin >> seeLoop)) {
            cout << "Invalid input; ";
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
        }
    }

    //password length checker
    cin.ignore();
    string pass;
    cout << endl << "Please enter a new password:" << endl;
    getline(cin, pass);
    while (pass.length() < 10) {
        if (pass.length() < 6)
            cout << "Password is weak, please try again." << endl;
        else if (pass.length() >= 6 && pass.length() <= 10)
            cout << "Password is medium, please try again." << endl;
        cin.ignore();
        getline(cin, pass);
    }
    cout << "Password is strong, you're good to go." << endl << endl;

    //traffic light simulator
    seeLoop = true;
    string color;
    while (seeLoop) {
        while (!(color == "red" || color == "green" || color == "yellow")) {
            cout << "What color is the light?" << endl;
            cin >> color;
            for (int i = 0; i < color.length(); i++) {
                color[i] = static_cast<char>(tolower(color[i]));
            }
        }
        if (color == "red")
            cout << "You should brake!" << endl;
        if (color == "yellow")
            cout << "You should slow down!" << endl;
        if (color == "green") {
            cout << "You should go!" << endl;
        }
        color = "";
        while ((cout << "Are there more lights? 1 for yes, 0 for no:" << endl)
         && !(cin >> seeLoop)) {
            cout << "Invalid input; ";
            cin.clear();
            cin.ignore(numeric_limits<streamsize>::max(), '\n');
         }
    }

    return 0;
}
