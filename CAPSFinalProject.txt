class issue {
    public:
    string name;
    int severity;
    string bodyText;

    issue(string n, int s, string t) {
        name = n;
        severity = s;
        bodyText = t;
    }

    issue();

    void print() {
        cout  << name << " - ";
        if (severity == 3) {
            cout << "High Severity" << endl;
        }
        else if (severity == 2) {
            cout << "Moderate Severity" << endl;
        }
        else if (severity == 1) {
            cout << "Low Severity" << endl;
        }
        cout << bodyText << endl;
    }

    void adjustSev(int s) {
        severity = s;
    }
};

int intInput() {
    int input;
    while (!(cin >> input)) {
        cout << "Invalid input, please try again." << endl;
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
    }
    if (input < 0) {
        cout << "Invalid input, please try again." << endl;
        input = intInput();
    }
    return input;
}

//obtains an int from 1 to 3, intended for the severity value
int sevInput() {
    int input;
    while (!(cin >> input)) {
        cout << "Invalid input, please try again." << endl;
        cin.clear();
        cin.ignore(numeric_limits<streamsize>::max(), '\n');
    }
    if (input != 1 && input != 2 && input != 3) {
        cout << "Invalid input, please try again." << endl;
        input = sevInput();
    }
    return input;
}

//adds a custom issue
void addIssue(int severity, string name, string bodyText) {
    if (severity == 3) {
        majIssues.push_back(issue{ name, severity, bodyText });
    }
    else if (severity == 2) {
        Issues.push_back(issue{ name, severity, bodyText });
    }
    else if (severity == 1) {
        minIssues.push_back(issue{ name, severity, bodyText });
    }
}

//removes an issue
void removeIssue(int severity, int index) {
    if (severity == 3 && index < majIssues.size()) {
        majIssues.erase(majIssues.begin() + index);
    }
    else if (severity == 2 && index < Issues.size()) {
        Issues.erase(Issues.begin() + index);
    }
    else if (severity == 1 && index < minIssues.size()) {
        minIssues.erase(minIssues.begin() + index);
    }
    else {
        cout << "Error" << endl;
    }
}

//fully prints an issue of a specific severity
void print(int severity) {
    if (severity == 3) {
        for (issue i : majIssues) {
            i.print();
        }
    }
    if (severity == 2) {
        for (issue i : Issues) {
            i.print();
        }
    }
    if (severity == 1) {
        for (issue i : minIssues) {
            i.print();
        }
    }
}

//gives a random topic of a specific severity
void randomTopic(int severity) {
    if (severity == 3) {
        int i = rand() % majIssues.size();
        majIssues.at(i).print();
    }
    if (severity == 2) {
        int i = rand() % Issues.size();
        Issues.at(i).print();
    }
    if (severity == 1) {
        int i = rand() % minIssues.size();
        minIssues.at(i).print();
    }
}


//returns a vector of issues that has find within the name
vector<issue> search(const string& find) {
    vector<issue> result;
    for (const issue& i : majIssues) {
        string n = i.name;
        std::transform(n.begin(), n.end(), n.begin(), ::tolower);
        if (n.find(find) != string::npos) {
            result.push_back(i);
        }
    }
    for (const issue& i : Issues) {
        string n = i.name;
        std::transform(n.begin(), n.end(), n.begin(), ::tolower);
        if (n.find(find) != string::npos) {
            result.push_back(i);
        }
    }
    for (const issue& i : minIssues) {
        string n = i.name;
        std::transform(n.begin(), n.end(), n.begin(), ::tolower);
        if (n.find(find) != string::npos) {
            result.push_back(i);
        }
    }
    return result;
}
